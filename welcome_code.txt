/* ================= TIKTOK WELCOME INTRO ENGINE ================= */
const welcomeIntro = document.getElementById("welcomeIntro");
let isWelcomeDismissed = false;

function dismissWelcomeIntro() {
  if (isWelcomeDismissed || !welcomeIntro) return;
  isWelcomeDismissed = true;

  // Kích hoạt animation trượt vút lên trên
  welcomeIntro.classList.add("hide-intro");

  // Tự động kích hoạt âm thanh (nếu có hàm phát nhạc)
  if (typeof startAudioPlayback === "function" && !isAudioPlaying) {
    startAudioPlayback();
  }

  // Gỡ bỏ hoàn toàn sau hiệu ứng transition 0.85s
  setTimeout(() => {
    welcomeIntro.style.display = "none";
  }, 850);
}

if (welcomeIntro) {
  // 1. Nhấp hoặc chạm bất kỳ đâu để vào ngay và mở nhạc
  welcomeIntro.addEventListener("click", () => {
    dismissWelcomeIntro();
  });

  welcomeIntro.addEventListener("touchstart", () => {
    dismissWelcomeIntro();
  }, { passive: true });

  // 2. Tự động trượt lên vào trang sau 3.8s (đủ thời gian xem trọn animation)
  const autoDismissTimer = setTimeout(() => {
    dismissWelcomeIntro();
  }, 3800);

  // 3. Hỗ trợ bấm phím Enter hoặc Space để vào trang
  window.addEventListener("keydown", (e) => {
    if (!isWelcomeDismissed && (e.key === "Enter" || e.key === " " || e.key === "Escape")) {
      clearTimeout(autoDismissTimer);
      dismissWelcomeIntro();
    }
  });
}
